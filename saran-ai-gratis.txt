Yang BENAR-BENAR gratis selamanya (tanpa kartu kredit):

1. GEMINI (sudah ada) - FREE PERMANEN
   - Endpoint: https://generativelanguage.googleapis.com/v1beta/models/...
   - Limit: ~15 RPM / 1500 RPD (Flash-Lite), reset tengah malam PT
   - Key: gratis dari aistudio.google.com
   - Fungsi: geminiChat() di lib/ai.js (sudah ada)

2. GROQ (sudah ada) - FREE PERMANEN
   - Endpoint: https://api.groq.com/openai/v1/chat/completions
   - Limit: 30 RPM / 1000 RPD / 8K TPM / 200K TPD per model
   - Model: openai/gpt-oss-20b, qwen/qwen3.8-27b, llama-4-scout
   - Fungsi: groqChat() di lib/ai.js (sudah ada)

3. POLLINATIONS (keyless, anti-mati) - GRATIS SELAMANYA TANPA DAFTAR
   - Endpoint: POST https://text.pollinations.ai/openai
   - Limit anon: 1 req / 15 detik. Daftar gratis: 1 req / 5 detik
   - Key: TIDAK BUTUH
   - Model: openai, mistral, deepseek, gemma, llama, qwen-coder
   - Catatan: rate ketat, model bisa gonta-ganti, tanpa SLA, timeout 30s

async function pollinationsChat(prompt, style = config.STYLE, signal) {
  const res = await fetch('https://text.pollinations.ai/openai', {
    method: 'POST',
    signal: signal || AbortSignal.timeout(30000),
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      model: 'openai',
      messages: [
        { role: 'system', content: style || config.STYLE },
        { role: 'user', content: String(prompt || '') }
      ],
      temperature: 0.7,
      max_tokens: 1024
    })
  });
  const json = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error('Pollinations error: ' + (json?.error?.message || res.status));
  const text = json?.choices?.[0]?.message?.content?.trim();
  if (!text) throw new Error('Pollinations respon kosong');
  return text;
}

4. OPENROUTER :free (cadangan, 50/hari) - GRATIS PERMANEN TANPA KARTU
   - Endpoint: https://openrouter.ai/api/v1/chat/completions
   - Limit: 20 RPM / 50 RPD (naik 1000 RPD jika pernah top-up $10, saldo tidak dipotong untuk :free)
   - Key: gratis dari openrouter.ai (wajib header HTTP-Referer + X-Title)
   - Model: qwen/qwen3-32b:free, deepseek/deepseek-r1:free, llama-3.3-70b-instruct:free

async function openRouterFreeChat(prompt, style = config.STYLE, signal) {
  if (!config.OPENROUTER_API_KEY) throw new Error('OPENROUTER_API_KEY kosong');
  const res = await fetch('https://openrouter.ai/api/v1/chat/completions', {
    method: 'POST',
    signal: signal || AbortSignal.timeout(20000),
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${config.OPENROUTER_API_KEY}`,
      'HTTP-Referer': 'https://wa-ai-bot-b',
      'X-Title': 'wa-ai-bot-b'
    },
    body: JSON.stringify({
      model: 'qwen/qwen3-32b:free',
      messages: [
        { role: 'system', content: style || config.STYLE },
        { role: 'user', content: String(prompt || '') }
      ],
      temperature: 0.7,
      max_tokens: 1024
    })
  });
  const json = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error('OpenRouter error: ' + (json?.error?.message || res.status));
  const text = json?.choices?.[0]?.message?.content?.trim();
  if (!text) throw new Error('OpenRouter respon kosong');
  return text;
}
